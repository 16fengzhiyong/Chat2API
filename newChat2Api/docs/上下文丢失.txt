### 方案A：前端导出为PDF（推荐用于纯前端项目）

使用 `jsPDF` + `html2canvas` 库实现报告导出 [[12]]：

```javascript
// 示例：导出测试结果报告
import jsPDF from 'jspdf';
import html2canvas from 'html2canvas';

function exportReport() {
  const reportElement = document.getElementById('result-report');
  html2canvas(reportElement).then(canvas => {
    const imgData = canvas.toDataURL('image/png');
    const pdf = new jsPDF('p', 'mm', 'a4');
    pdf.addImage(imgData, 'PNG', 0, 0, 210, 297);
    pdf.save('big-five-report.pdf');
  });
}
```

```javascript
// 保存测试结果
function saveResults(results) {
  const savedTests = JSON.parse(localStorage.getItem('bigFiveTests') || '[]');
  savedTests.push({
    timestamp: new Date().toISOString(),
    scores: results
  });
  localStorage.setItem('bigFiveTests', JSON.stringify(savedTests));
}
```
